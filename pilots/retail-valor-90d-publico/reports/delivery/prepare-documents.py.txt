def sha(q):return hashlib.sha256(q.read_bytes()).hexdigest()
def get(rel,field):
 o=json.loads((p/rel).read_text())
 for part in field.split('.'):o=o[int(part)] if isinstance(o,list) else o[part]
 return o

def fmt(x,style):
 if style=='money':return 'GBP '+f'{x:,.2f}'.replace(',','@').replace('.',',').replace('@','.')
 if style=='percent_fraction':return f'{100*x:.2f}'.replace('.',',')+'%'
 if style=='percent':return f'{x:.2f}'.replace('.',',')+'%'
 if style=='integer':return f'{x:,}'.replace(',','.')
 if style=='date':return str(x)[:10]
 if style=='decimal3':return f'{x:.3f}'.replace('.',',')
 return str(x)
